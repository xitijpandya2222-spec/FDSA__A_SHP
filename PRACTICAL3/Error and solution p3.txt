3.1 ----> Errors in all sorting method but solved .
	
	1. Bubble sort 

	for(int i = 0; i < n - 1; i++)
    {
        for(int j = 0; j < n - 1 - i; j++)
        {
            if(arr[j] > arr[j + 1])
            {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }

	2. Selection sort 

	 for(int i = 0; i < n - 1; i++)
    {
        int lowest = i;

        for(int j = i + 1; j < n; j++)
        {
            if(arr[j] < arr[lowest])
            {
                lowest = j;
            }
        }

        int temp = arr[i];
        arr[i] = arr[lowest];
        arr[lowest] = temp;
    }

	3. Insertion sort 

	for(int i = 1; i < n; i++)
    {
        int current = arr[i];
        int j = i - 1;
        while(j >= 0 && arr[j] > current)
        {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = current;
    }
